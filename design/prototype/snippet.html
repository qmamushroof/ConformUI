DOM snippet

Run this script on your own page to copy a DOM snapshot: real element boxes, text and styles, so measured checks report exact pixel values and selectors.

  1. Open your page at the capture width

    In Chrome, open the page you captured. Use 1440 px for Desktop and 390 px for Mobile; the DevTools device toolbar sets an exact width.

  2. Paste the snippet into the DevTools Console and press Enter

    Open the Console with Ctrl+Shift+J (Cmd+Option+J on Mac). It prints ConformUI snapshot copied: 214 elements, 1440×1000. If Chrome asks you to type "allow pasting", type it, then paste again.

  3. Come back and choose Paste snapshot

    In your project, under the same viewport, choose Paste snapshot. Open Sample landing page

/* ConformUI DOM snippet 1.0.0
   Reads layout, text and styles only; sends nothing. */
(() => {
  const LANDMARK = 'header,nav,main,footer,aside,section[aria-label]';
  const PICK = LANDMARK + ',h1,h2,h3,h4,h5,h6,a[href],button,[role=button],' +
    'img,svg[role=img],input:not([type=hidden]),select,textarea,p,li,label';
  const kind = (el) => el.matches(LANDMARK) ? 'landmark' : /^H\d$/.test(el.tagName) ? 'heading'
    : el.matches('a[href]') ? 'link' : el.matches('button,[role=button]') ? 'button'
    : el.matches('img,svg') ? 'image' : el.matches('input,select,textarea') ? 'control' : 'text';
  const sel = (el) => {
    const path = [];
    for (let n = el; n && n !== document.body && path.length < 5; n = n.parentElement) {
      if (n.id && !/\d{4,}|[a-f0-9]{8,}/.test(n.id)) { path.unshift('#' + CSS.escape(n.id)); break; }
      const same = [...n.parentElement.children].filter((c) => c.tagName === n.tagName);
      path.unshift(n.tagName.toLowerCase() + (same.length > 1 ? `:nth-of-type(${same.indexOf(n) + 1})` : ''));
    }
    return path.join(' > ');
  };
  const px = (v) => parseFloat(v) || 0, r2 = (v) => Math.round(v * 100) / 100;
  const doc = document.documentElement;
  const elements = [...document.querySelectorAll(PICK)].slice(0, 1500).flatMap((el) => {
    const r = el.getBoundingClientRect(), s = getComputedStyle(el);
    if (!r.width || !r.height || s.display === 'none' || s.visibility === 'hidden') return [];
    const isControl = el.matches('input,select,textarea');
    return [{ kind: kind(el), tag: el.tagName.toLowerCase(), selector: sel(el),
      name: (el.getAttribute('aria-label') || el.getAttribute('alt') || '').slice(0, 120),
      text: isControl ? undefined : el.innerText.replace(/\s+/g, ' ').trim().slice(0, 200),
      box: { x: r2(r.left + scrollX), y: r2(r.top + scrollY), w: r2(r.width), h: r2(r.height) },
      style: { fontSize: px(s.fontSize), fontWeight: px(s.fontWeight), color: s.color,
        backgroundColor: s.backgroundColor, marginTop: px(s.marginTop), borderRadius: px(s.borderRadius) },
      flags: { beyondViewport: r.right > doc.clientWidth + 1 || r.left < -1 } }];
  }).map((e, i) => ({ i, ...e }));
  const json = JSON.stringify({ schema: 'conformui.dom-snapshot', version: 1, snippetVersion: '1.0.0',
    page: { url: location.origin + location.pathname, title: document.title.slice(0, 120) },
    capturedAt: new Date().toISOString(), viewport: { width: innerWidth, height: innerHeight },
    dpr: devicePixelRatio, document: { scrollWidth: doc.scrollWidth, clientWidth: doc.clientWidth },
    overflow: { horizontal: doc.scrollWidth > doc.clientWidth, horizontalPx: Math.max(0, doc.scrollWidth - doc.clientWidth) },
    elements, truncated: elements.length >= 1500 });
  typeof copy === 'function' ? copy(json) : console.log(json);
  return `ConformUI snapshot copied: ${elements.length} elements, ${innerWidth}×${innerHeight}`;
})();

Reads layout, text and styles only; sends nothing

What the snapshot containsUp to 1500 elements, 1 MB

Reads

  • Page origin and path, title, viewport size and pixel density
  • Landmarks, headings, links, buttons, images, form controls and text blocks
  • Each element's box, a stable selector, its accessible name and up to 200 characters of text
  • Computed font size and weight, colours, margins and radius
  • Whether the page scrolls sideways, and by how much

Never reads

  • Form values, checked state or hidden inputs
  • Cookies, localStorage, sessionStorage or IndexedDB
  • Query strings or URL fragments
  • Anything over the network: the script makes no requests
